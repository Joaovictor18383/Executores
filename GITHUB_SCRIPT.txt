@echo off
title Madium Manager [Sky land]
setlocal enabledelayedexpansion

:: Verifica privilégios de administrador
net session >nul 2>&1
if %errorlevel% neq 0 (
    echo ==================================================
    echo    SOLICITANDO PRIVILEGIOS DE ADMINISTRADOR...
    echo ==================================================
    powershell -NoProfile -Command "Start-Process '%~f0' -Verb RunAs"
    exit /b
)

:menu
cls
echo ==================================================
echo              MADIUM MANAGER - MENU
echo ==================================================
echo [1] Madium
echo [2] Deletar arquivos do Madium (Limpeza)
echo [3] Baixar e Instalar Dependencias
echo [4] Baixar e Instalar Fishstrap
echo [5] Sair
echo ==================================================
set /p opt="Escolha uma opcao: "

if "%opt%"=="1" goto nova_versao_key
if "%opt%"=="2" goto deletar_madium
if "%opt%"=="3" goto dependencias
if "%opt%"=="4" goto baixar_fishstrap
if "%opt%"=="5" exit
goto menu

:nova_versao_key
cls
echo ==================================================
echo         NOVA VERSAO DO MADIUM MAIS TEM KEY
echo ==================================================
set "inst_url=https://files.manuscdn.com/user_upload_by_module/session_file/310519663151269258/KMFVNDQmHinAehOS.exe"
set "inst_path=%temp%\Madium_Installer_Key.exe"

echo Baixando instalador...
curl -L -o "%inst_path%" "%inst_url%"

if not exist "%inst_path%" echo [!] Erro no download do instalador. && pause && goto menu

echo Iniciando instalacao...
start /wait "" "%inst_path%"

echo Limpando arquivos temporarios...
del "%inst_path%" /f /q >nul 2>&1

echo.
echo Operacao concluida.
pause
goto menu

:deletar_madium
cls
echo Limpando arquivos do Madium...
taskkill /F /IM Madium-Bootstrapper.exe >nul 2>&1
taskkill /F /IM Madium.exe >nul 2>&1
taskkill /F /IM madium.exe >nul 2>&1
taskkill /F /IM Loader.exe >nul 2>&1
timeout /t 1 /nobreak >nul
rmdir /S /Q "%LocalAppData%\Madium" >nul 2>&1
echo Limpeza concluida.
pause
goto menu

:dependencias
cls
echo ==================================================
echo             VERIFICANDO DEPENDENCIAS
echo ==================================================

set "dotnet8_installed=0"
where dotnet >nul 2>&1
if %errorlevel% equ 0 (
    powershell -Command "dotnet --list-runtimes" | findstr /i "Microsoft.WindowsDesktop.App 8.0" >nul
    if !errorlevel! equ 0 set "dotnet8_installed=1"
)
if %dotnet8_installed% equ 0 if exist "%ProgramFiles%\dotnet\shared\Microsoft.WindowsDesktop.App\8.0*" set "dotnet8_installed=1"

set "vcredist64_installed=0"
reg query "HKLM\SOFTWARE\Microsoft\VisualStudio\14.0\VC\Runtimes\x64" /v "Installed" >nul 2>&1 && set "vcredist64_installed=1"
if %vcredist64_installed% equ 0 if exist "%SystemRoot%\System32\vcruntime140.dll" set "vcredist64_installed=1"

set "vcredist86_installed=0"
reg query "HKLM\SOFTWARE\WOW6432Node\Microsoft\VisualStudio\14.0\VC\Runtimes\x86" /v "Installed" >nul 2>&1 && set "vcredist86_installed=1"
if %vcredist86_installed% equ 0 if exist "%SystemRoot%\SysWOW64\vcruntime140.dll" set "vcredist86_installed=1"

set "webview2_installed=0"
reg query "HKLM\SOFTWARE\WOW6432Node\Microsoft\EdgeUpdate\Clients\{F3C4FE00-EFC5-498D-9529-EE5816BB8851}" /v "pv" >nul 2>&1 && set "webview2_installed=1"
if %webview2_installed% equ 0 if exist "%ProgramFiles(x86)%\Microsoft\EdgeWebView\Application" set "webview2_installed=1"

echo [1/4] .NET Runtime 8.0: 
if %dotnet8_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\dotnet8.exe" "https://builds.dotnet.microsoft.com/dotnet/WindowsDesktop/8.0.27/windowsdesktop-runtime-8.0.27-win-x64.exe"
    start /wait "" "%temp%\dotnet8.exe" /install /quiet /norestart
    del "%temp%\dotnet8.exe" /f /q >nul 2>&1
)

echo [2/4] Visual C++ x64:
if %vcredist64_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\vc64.exe" "https://aka.ms/vc14/vc_redist.x64.exe"
    start /wait "" "%temp%\vc64.exe" /install /quiet /norestart
    del "%temp%\vc64.exe" /f /q >nul 2>&1
)

echo [3/4] Visual C++ x86:
if %vcredist86_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\vc86.exe" "https://aka.ms/vc14/vc_redist.x86.exe"
    start /wait "" "%temp%\vc86.exe" /install /quiet /norestart
    del "%temp%\vc86.exe" /f /q >nul 2>&1
)

echo [4/4] WebView2:
if %webview2_installed% equ 1 (echo    - JA INSTALADO.) else (
    echo    - BAIXANDO...
    curl -L -o "%temp%\wv2.exe" "https://go.microsoft.com/fwlink/?linkid=2124701"
    start /wait "" "%temp%\wv2.exe" /silent /install
    del "%temp%\wv2.exe" /f /q >nul 2>&1
)

echo.
echo Operacao concluida.
pause
goto menu

:baixar_fishstrap
cls
echo ==================================================
echo            BAIXANDO FISHSTRAP E DEPENDENCIAS
echo ==================================================

echo [1/2] Baixando .NET 6.0 Desktop Runtime...
curl -L -o "%temp%\dotnet6.exe" "https://builds.dotnet.microsoft.com/dotnet/WindowsDesktop/6.0.36/windowsdesktop-runtime-6.0.36-win-x64.exe"
if exist "%temp%\dotnet6.exe" start /wait "" "%temp%\dotnet6.exe" /install /quiet /norestart && del "%temp%\dotnet6.exe" /f /q >nul 2>&1

echo [2/2] Baixando Fishstrap...
curl -L -o "%temp%\Fishstrap.exe" "https://fishstrap.app/Fishstrap.exe"
if exist "%temp%\Fishstrap.exe" start "" "%temp%\Fishstrap.exe" && timeout /t 10 /nobreak >nul && del "%temp%\Fishstrap.exe" /f /q >nul 2>&1

echo.
echo Operacao concluida.
pause
goto menu
