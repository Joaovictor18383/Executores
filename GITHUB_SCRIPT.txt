function Set-Pos {

    param(
        [int]$X,
        [int]$Y
    )

    try {
        $pos = New-Object System.Management.Automation.Host.Coordinates($X,$Y)
        $Host.UI.RawUI.CursorPosition = $pos
    } catch {}
}


function Center-Line {

    param(
        [string]$Text = "",
        [ConsoleColor]$Color = "White"
    )

    $pad = [Math]::Max(
        0,
        [Math]::Floor(($ConsoleWidth - $Text.Length) / 2)
    )

    Write-Host (" " * $pad) -NoNewline
    Write-Host $Text -ForegroundColor $Color
}


function Title-Line {

    param(
        [string]$Sky,
        [string]$Land
    )

    $total = $Sky.Length + $Land.Length

    $pad = [Math]::Max(
        0,
        [Math]::Floor(($ConsoleWidth - $total) / 2)
    )

    Write-Host (" " * $pad) -NoNewline
    Write-Host $Sky -NoNewline -ForegroundColor White
    Write-Host $Land -ForegroundColor Magenta
}

function Download-File {

    param(
        [string]$Url,
        [string]$Destination
    )

    try {

        if (Test-Path $Destination) {
            Remove-Item -LiteralPath $Destination -Force -ErrorAction SilentlyContinue
        }

        & curl.exe `
            -f `
            -L `
            --retry 3 `
            --retry-delay 2 `
            -o $Destination `
            $Url

        if ($LASTEXITCODE -ne 0) {
            throw "Falha no download."
        }

        if (-not (Test-Path $Destination)) {
            throw "Arquivo nao foi criado."
        }

        $file = Get-Item -LiteralPath $Destination -ErrorAction Stop

        if ($file.Length -le 0) {
            throw "Arquivo vazio."
        }

        return $true

    } catch {

        Remove-Item `
            -LiteralPath $Destination `
            -Force `
            -ErrorAction SilentlyContinue

        return $false
    }
}

function Test-WebView2 {

    if (
        (Test-Path "${env:ProgramFiles(x86)}\Microsoft\EdgeWebView\Application") -or
        (Test-Path "$env:ProgramFiles\Microsoft\EdgeWebView\Application")
    ) {
        return $true
    }

    return $false
}


function Install-WebView2 {

    Show-Working "Verificando Microsoft Edge WebView2..."

        return $true
    }

    $installer = Join-Path $env:TEMP "MicrosoftEdgeWebView2.exe"

        Show-Error "Falha ao baixar WebView2."

        return $false
    }

    Show-Working "Instalando WebView2..."

    try {

        Start-Process `
            -FilePath $installer `
            -ArgumentList "/silent /install" `
            -Wait

        Remove-Item `
            -LiteralPath $installer `
            -Force `
            -ErrorAction SilentlyContinue

        Show-Success "WebView2 instalado."

        return $true

    } catch {

        Remove-Item `
            -LiteralPath $installer `
            -Force `
            -ErrorAction SilentlyContinue

        Show-Error "Falha ao instalar WebView2."

        return $false
    }

function Test-DotNetDesktop {

    param(
        [string]$Major
    )

    try {

        $dotnet = Get-Command dotnet -ErrorAction Stop
        $runtimes = & $dotnet.Source --list-runtimes 2>$null

        foreach ($runtime in $runtimes) {

            if ($runtime -match "Microsoft\.WindowsDesktop\.App $Major\.") {
                return $true
            }
        }

    } catch {}

    return $false
}

function Test-VCRuntime {

    param(
        [ValidateSet("x64","x86")]
        [string]$Architecture
    )

    $paths = @()

    if ($Architecture -eq "x64") {
        $paths += "HKLM:\SOFTWARE\Microsoft\VisualStudio\14.0\VC\Runtimes\x64"
    }
    else {
        $paths += "HKLM:\SOFTWARE\WOW6432Node\Microsoft\VisualStudio\14.0\VC\Runtimes\x86"
        $paths += "HKLM:\SOFTWARE\Microsoft\VisualStudio\14.0\VC\Runtimes\x86"
    }

    foreach ($path in $paths) {
        try {
            $value = Get-ItemProperty -Path $path -Name Installed -ErrorAction Stop
            if ($value.Installed -eq 1) {
                return $true
            }
        } catch {}
    }

    return $false
}


function Install-VCRuntime {

    param(
        [ValidateSet("x64","x86")]
        [string]$Architecture
    )

    if (Test-VCRuntime $Architecture) {
        Show-Success "Visual C++ $Architecture ja esta instalado."
        return $true
    }

    if ($Architecture -eq "x64") {
        $url = $VC64_URL
        $installer = Join-Path $env:TEMP "vc_redist.x64.exe"
    }
    else {
        $url = $VC86_URL
        $installer = Join-Path $env:TEMP "vc_redist.x86.exe"
    }

    Show-Working "Baixando Visual C++ $Architecture..."

    if (-not (Download-File $url $installer)) {
        Show-Error "Falha ao baixar Visual C++ $Architecture."
        return $false
    }

    Show-Working "Instalando Visual C++ $Architecture..."

    try {

        $process = Start-Process `
            -FilePath $installer `
            -ArgumentList "/install /passive /norestart" `
            -Verb RunAs `
            -Wait `
            -PassThru

        $exitCode = $process.ExitCode

        Remove-Item `
            -LiteralPath $installer `
            -Force `
            -ErrorAction SilentlyContinue

        if ($exitCode -eq 0 -or $exitCode -eq 1638 -or $exitCode -eq 3010) {
            Show-Success "Visual C++ $Architecture instalado/verificado."
            return $true
        }

        Show-Error "Visual C++ $Architecture retornou codigo $exitCode."
        return $false

    } catch {

        Remove-Item `
            -LiteralPath $installer `
            -Force `
            -ErrorAction SilentlyContinue

        Show-Error "Falha ao instalar Visual C++ $Architecture."
        return $false
    }
}

# ============================================================================
# FISHSTRAP
# ============================================================================

function Download-Fishstrap {

    param(
        [string]$ExecutorName
    )

    Show-Action `
        "$ExecutorName - FISHSTRAP" `
        "STATUS: PREPARANDO FISHSTRAP..."

    if (-not (Install-DotNet6)) {
        Pause-SkyLand
        return
    }

    $destination = Join-Path $Downloads "Fishstrap.exe"

    try {
        Get-Process -Name "Fishstrap" -ErrorAction SilentlyContinue |
            Stop-Process -Force -ErrorAction SilentlyContinue
    } catch {}

    Show-Working "Procurando versao mais recente do Fishstrap..."

    try {

        $release = Invoke-RestMethod `
            -Uri "https://api.github.com/repos/fishstrap/fishstrap/releases/latest" `
            -Headers @{
                "User-Agent" = "SKY-LAND"
                "Accept" = "application/vnd.github+json"
            } `
            -ErrorAction Stop

        $asset = $release.assets |
            Where-Object {
                $_.name -ieq "Fishstrap.exe"
            } |
            Select-Object -First 1

        if (-not $asset) {
            throw "Fishstrap.exe nao encontrado na release mais recente."
        }

        $downloadUrl = $asset.browser_download_url

        if ([string]::IsNullOrWhiteSpace($downloadUrl)) {
            throw "Link de download do Fishstrap nao encontrado."
        }

        Show-Working "Baixando Fishstrap $($release.tag_name)..."

        if (-not (Download-File $downloadUrl $destination)) {
            throw "Falha no download do Fishstrap."
        }

        Show-Success "Fishstrap $($release.tag_name) baixado."

        Show-Working "Executando Fishstrap..."

        Start-Process -FilePath $destination

        Show-Success "Fishstrap iniciado."

    } catch {

        Show-Error "Falha ao baixar Fishstrap."

        if ($_.Exception.Message) {
            Write-Host ""
            Center-Line $_.Exception.Message DarkGray
        }

        Pause-SkyLand
        return
    }

    Pause-SkyLand
}


# ============================================================================
# ROBLOX ACCOUNT MANAGER
# ============================================================================

function Download-RobloxAccountManager {

    param(
        [string]$ExecutorName
    )

    Show-Action `
        "$ExecutorName - ROBLOX ACCOUNT MANAGER" `
        "STATUS: PROCURANDO VERSAO MAIS RECENTE..."

    $zip = Join-Path $env:TEMP "RobloxAccountManager.zip"
    $folder = Join-Path $Downloads "Roblox Account Manager"

    Remove-Item `
        -LiteralPath $zip `
        -Force `
        -ErrorAction SilentlyContinue

    try {

        $release = Invoke-RestMethod `
            -Uri "https://api.github.com/repos/ic3w0lf22/Roblox-Account-Manager/releases/latest" `
            -Headers @{
                "User-Agent" = "SKY-LAND"
            } `
            -ErrorAction Stop

        $asset = $release.assets |
            Where-Object {
                $_.name -like "*.zip"
            } |
            Select-Object -First 1

        if (-not $asset) {
            throw "ZIP nao encontrado."
        }

        Show-Working "Baixando Roblox Account Manager..."

        if (-not (Download-File $asset.browser_download_url $zip)) {
            throw "Falha no download."
        }

        if (Test-Path $folder) {

            Remove-Item `
                -LiteralPath $folder `
                -Recurse `
                -Force `
                -ErrorAction SilentlyContinue
        }

        New-Item `
            -ItemType Directory `
            -Path $folder `
            -Force |
            Out-Null

        Show-Working "Extraindo arquivos..."

        Expand-Archive `
            -LiteralPath $zip `
            -DestinationPath $folder `
            -Force

        Remove-Item `
            -LiteralPath $zip `
            -Force `
            -ErrorAction SilentlyContinue

        Show-Success "Roblox Account Manager baixado."

    } catch {

        Show-Error "Falha ao baixar Roblox Account Manager."
    }

    Pause-SkyLand
}
