@echo off
chcp 65001 >nul
title SKY LAND - Atualizador do GitHub
color 0B

echo ==========================================
echo       SKY LAND - ATUALIZADOR GITHUB
echo ==========================================
echo.

where git >nul 2>&1
if errorlevel 1 (
    echo ERRO: Git nao encontrado!
    echo Instale o Git ou o GitHub Desktop.
    pause
    exit /b 1
)

echo Selecione o arquivo ZIP atualizado...
echo.

powershell -NoProfile -Command "Add-Type -AssemblyName System.Windows.Forms; $f=New-Object System.Windows.Forms.OpenFileDialog; $f.Filter='Arquivos ZIP (*.zip)|*.zip'; $f.Title='Selecione o ZIP do SKY LAND'; if($f.ShowDialog() -eq 'OK'){Set-Content -Path '%temp%\skyland_zip.txt' -Value $f.FileName -Encoding UTF8}"

if not exist "%temp%\skyland_zip.txt" (
    echo Nenhum arquivo selecionado.
    pause
    exit /b 1
)

set /p ZIP=<"%temp%\skyland_zip.txt"
del "%temp%\skyland_zip.txt"

echo.
echo Selecione a pasta do repositorio SKY LAND...
echo.

powershell -NoProfile -Command "Add-Type -AssemblyName System.Windows.Forms; $f=New-Object System.Windows.Forms.FolderBrowserDialog; $f.Description='Selecione a pasta do repositorio SKY LAND'; if($f.ShowDialog() -eq 'OK'){Set-Content -Path '%temp%\skyland_repo.txt' -Value $f.SelectedPath -Encoding UTF8}"

if not exist "%temp%\skyland_repo.txt" (
    echo Nenhuma pasta selecionada.
    pause
    exit /b 1
)

set /p REPO=<"%temp%\skyland_repo.txt"
del "%temp%\skyland_repo.txt"

if not exist "%REPO%\.git" (
    echo ERRO: A pasta selecionada nao e um repositorio Git.
    pause
    exit /b 1
)

echo.
echo Extraindo arquivos...

set "TEMP_DIR=%TEMP%\SKYLAND_UPDATE"

if exist "%TEMP_DIR%" rmdir /s /q "%TEMP_DIR%"
mkdir "%TEMP_DIR%"

powershell -NoProfile -Command "Expand-Archive -LiteralPath '%ZIP%' -DestinationPath '%TEMP_DIR%' -Force"

if errorlevel 1 (
    echo ERRO ao extrair o ZIP.
    pause
    exit /b 1
)

echo.
echo Copiando arquivos para o repositorio...

robocopy "%TEMP_DIR%" "%REPO%" /E /XD .git bin obj .vs /XF *.user *.suo

echo.
echo Enviando atualizacoes para o GitHub...

cd /d "%REPO%"

git add -A
git commit -m "Atualizacao do SKY LAND"
if errorlevel 1 (
    echo.
    echo Nenhuma alteracao nova para enviar ou erro no commit.
    pause
    exit /b 1
)

git push origin main

if errorlevel 1 (
    echo.
    echo ERRO ao enviar para o GitHub.
    pause
    exit /b 1
)

echo.
echo ==========================================
echo       ATUALIZACAO CONCLUIDA!
echo ==========================================
echo.
echo Os arquivos foram enviados para o GitHub.
echo.
pause
