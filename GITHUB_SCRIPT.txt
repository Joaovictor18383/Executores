function Open-Potassium {
    while ($true) {
        Show-ExecutorMenu "POTASSIUM" "BAIXAR POTASSIUM"

        Set-Pos 2 36
        Write-Host "Escolha uma opcao: " -ForegroundColor White -NoNewline
        $op = Read-Host

        switch ($op) {
            "1" { Download-Potassium }
            "2" { Install-PotassiumDependencies }
            "3" { Download-Fishstrap "POTASSIUM" }
            "4" { Download-RobloxAccountManager "POTASSIUM" }
            "5" { Download-MultiBloxy "POTASSIUM" }
            "6" { return }
            default {
                Set-Pos 2 38
                Write-Host "OPCAO INVALIDA!" -ForegroundColor Magenta
                Start-Sleep -Seconds 1
            }
        }
    }
}

# STATUS DOS EXECUTORES - WEAO
# ============================================================================

function Check-ExecutorStatus {
    Show-Action "OUTROS - STATUS DOS EXECUTORES" "STATUS: CONSULTANDO WEAO..."

    Write-Host ""
    Center-Line "Verificando Potassium, Madium e Real..." Magenta
    Write-Host ""

    try {
        $headers = @{
            "User-Agent" = "WEAO-3PService"
            "Accept" = "application/json"
        }

        $statuses = Invoke-RestMethod -Uri "https://weao.xyz/api/status/exploits" -Headers $headers -Method Get -TimeoutSec 15 -ErrorAction Stop

        function Show-WEAOStatus {
            param([string]$Name)

            $item = $statuses |
                Where-Object { $_.title -and $_.title.Trim().ToLower() -eq $Name.ToLower() } |
                Select-Object -First 1

            if (-not $item) {
                $item = $statuses |
                    Where-Object { $_.title -and $_.title.ToLower().Contains($Name.ToLower()) } |
                    Select-Object -First 1
            }

            if (-not $item) {
                Center-Line ("{0,-12} : NAO ENCONTRADO NO WEAO" -f $Name.ToUpper()) DarkGray
                return
            }

            if ($item.updateStatus -eq $true) {
                $state = "ON"
                $color = [ConsoleColor]::Green
            }
            else {
                $state = "OFF / DESATUALIZADO"
                $color = [ConsoleColor]::Red
            }

            $version = ""
            if ($item.version) {
                $version = "  |  VERSAO: $($item.version)"
            }

            Center-Line ("{0,-12} : {1}{2}" -f $Name.ToUpper(), $state, $version) $color
        }

        Show-WEAOStatus "Potassium"
        Write-Host ""
        Show-WEAOStatus "Madium"
        Write-Host ""
        Show-WEAOStatus "Real"

        Write-Host ""
        Center-Line "Fonte: WhatExpsAre.Online / WEAO API" DarkGray
    }
    catch {
        Show-Error "Nao foi possivel consultar o status dos executores."
        Write-Host ""
        Center-Line "Verifique sua internet ou tente novamente mais tarde." DarkGray
    }

    Pause-SkyLand
}

# ============================================================================
# ANYDESK
# ============================================================================

function Download-AnyDesk {
    Show-Action "OUTROS - ANYDESK" "STATUS: PREPARANDO DOWNLOAD..."

    $destination = Join-Path $Downloads "AnyDesk.exe"

    try {
        Get-Process -Name "AnyDesk" -ErrorAction SilentlyContinue |
            Stop-Process -Force -ErrorAction SilentlyContinue
    } catch {}

    Show-Working "Baixando AnyDesk..."

    if (Download-File $ANYDESK_URL $destination) {
        Show-Success "AnyDesk baixado com sucesso."
        Write-Host ""
        Center-Line $destination DarkGray
    }
    else {
        Show-Error "Falha ao baixar AnyDesk."
    }

    Pause-SkyLand
}
