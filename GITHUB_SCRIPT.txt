echo.
echo                                           VERIFICANDO DEPENDENCIAS

)

:baixar_fishstrap

echo                    ============================================================================
echo.
echo                                              BAIXANDO FISHSTRAP

echo                    ============================================================================
echo.

set "fishstrap_dir=%USERPROFILE%\Downloads\Fishstrap"
set "fishstrap_exe=%fishstrap_dir%\Fishstrap.exe"

if not exist "%fishstrap_dir%" mkdir "%fishstrap_dir%" >nul 2>&1

echo                                       [1/2] Verificando .NET Desktop Runtime 6...
echo.

set "dotnet6_installed=0"
where dotnet.exe >nul 2>&1
if not errorlevel 1 (
    for /f "delims=" %%R in ('dotnet --list-runtimes 2^>nul ^| findstr /I /C:"Microsoft.WindowsDesktop.App 6."') do set "dotnet6_installed=1"
)

if "%dotnet6_installed%"=="1" (
    echo                                       .NET 6 ja instalado.
) else (
    if exist "%temp%\dotnet6.exe" del /f /q "%temp%\dotnet6.exe" >nul 2>&1
    curl.exe -fLsS --retry 2 --connect-timeout 15 -o "%temp%\dotnet6.exe" "https://builds.dotnet.microsoft.com/dotnet/WindowsDesktop/6.0.36/windowsdesktop-runtime-6.0.36-win-x64.exe"
    if errorlevel 1 (
        echo.
        echo                                       [!] Erro ao baixar .NET 6.
        if exist "%temp%\dotnet6.exe" del /f /q "%temp%\dotnet6.exe" >nul 2>&1
        pause
        goto menu
    )
    if not exist "%temp%\dotnet6.exe" (
        echo.
        echo                                       [!] Instalador do .NET 6 nao encontrado.
        pause
        goto menu
    )
    start /wait "" "%temp%\dotnet6.exe" /install /quiet /norestart
    del /f /q "%temp%\dotnet6.exe" >nul 2>&1
)

echo.
echo                                       [2/2] Baixando Fishstrap...
echo.

if exist "%fishstrap_exe%" del /f /q "%fishstrap_exe%" >nul 2>&1

curl.exe -fLsS --retry 3 --connect-timeout 15 -o "%fishstrap_exe%" "https://github.com/fishstrap/fishstrap/releases/latest/download/Fishstrap.exe"

if errorlevel 1 (
    echo.
    echo                                       [!] Erro ao baixar Fishstrap.
    if exist "%fishstrap_exe%" del /f /q "%fishstrap_exe%" >nul 2>&1
    pause
    goto menu
)

if not exist "%fishstrap_exe%" (
    echo.
    echo                                       [!] Fishstrap nao encontrado.
    pause
    goto menu
)

for %%A in ("%fishstrap_exe%") do if %%~zA LEQ 0 (
    echo                                       [!] O arquivo baixado esta vazio.
    del /f /q "%fishstrap_exe%" >nul 2>&1
    pause
    goto menu
)

echo.
echo                                       Download concluido.
echo                                       Pasta:
echo                                       %fishstrap_dir%
echo.

start "" "%fishstrap_dir%"

pause
goto menu


:: ============================================================
:: ROBLOX ACCOUNT MANAGER
:: ============================================================

:baixar_ram
cls
echo.
echo                    ============================================================================
echo.
echo                                      BAIXANDO ROBLOX ACCOUNT MANAGER
echo.
echo                    ============================================================================
echo.

set "ram_zip=%temp%\RobloxAccountManager.zip"
set "ram_dir=%USERPROFILE%\Downloads\Roblox Account Manager"

if exist "%ram_zip%" del "%ram_zip%" /f /q >nul 2>&1

echo                                       Procurando versao mais recente...
echo.

powershell.exe -NoProfile -ExecutionPolicy Bypass -Command ^
"$ErrorActionPreference='Stop'; $ProgressPreference='SilentlyContinue'; ^
$r=Invoke-RestMethod 'https://api.github.com/repos/ic3w0lf22/Roblox-Account-Manager/releases/latest'; ^
$a=$r.assets | Where-Object { $_.name -like '*.zip' } | Select-Object -First 1; ^
if(-not $a){throw 'Nenhum ZIP encontrado na release mais recente'}; ^
Invoke-WebRequest -UseBasicParsing -Uri $a.browser_download_url -OutFile (Join-Path $env:TEMP 'RobloxAccountManager.zip')"

if errorlevel 1 (
    echo.
    echo                                       [!] Erro ao baixar Roblox Account Manager.
    if exist "%ram_zip%" del "%ram_zip%" /f /q >nul 2>&1
    pause
    goto menu
)

if not exist "%ram_zip%" (
    echo.
    echo                                       [!] ZIP do Roblox Account Manager nao encontrado.
    pause
    goto menu
)

for %%A in ("%ram_zip%") do if %%~zA LEQ 0 (
    echo                                       [!] O ZIP baixado esta vazio.
    del "%ram_zip%" /f /q >nul 2>&1
    pause
    goto menu
)

echo.
echo                                       Extraindo arquivos...
echo.

if not exist "%ram_dir%" mkdir "%ram_dir%" >nul 2>&1

powershell.exe -NoProfile -ExecutionPolicy Bypass -Command ^
"$ErrorActionPreference='Stop'; Expand-Archive -LiteralPath (Join-Path $env:TEMP 'RobloxAccountManager.zip') -DestinationPath (Join-Path $env:USERPROFILE 'Downloads\Roblox Account Manager') -Force"

if errorlevel 1 (
    echo.
    echo                                       [!] Erro ao extrair os arquivos.
    del "%ram_zip%" /f /q >nul 2>&1
    pause
    goto menu
)

del "%ram_zip%" /f /q >nul 2>&1

echo.
echo                                       Download concluido.
echo                                       A pasta existente foi preservada.
echo                                       Pasta:
echo                                       %ram_dir%
echo.

start "" "%ram_dir%"

pause
goto menu
