:: ============================================================
:: DEPENDENCIAS
:: ============================================================

:dependencias
cls
echo.
echo                    ============================================================================
echo.
echo                                           VERIFICANDO DEPENDENCIAS
echo.
echo                    ============================================================================
echo.

set "dotnet8_installed=0"

where dotnet >nul 2>&1

if %errorlevel% equ 0 (
    dotnet --list-runtimes | findstr /i "Microsoft.WindowsDesktop.App 8.0" >nul

    if !errorlevel! equ 0 (
        set "dotnet8_installed=1"
    )
)

set "vcredist64_installed=0"

reg query "HKLM\SOFTWARE\Microsoft\VisualStudio\14.0\VC\Runtimes\x64" /v "Installed" >nul 2>&1

if %errorlevel% equ 0 (
    set "vcredist64_installed=1"
)

set "vcredist86_installed=0"

reg query "HKLM\SOFTWARE\WOW6432Node\Microsoft\VisualStudio\14.0\VC\Runtimes\x86" /v "Installed" >nul 2>&1

if %errorlevel% equ 0 (
    set "vcredist86_installed=1"
)

set "webview2_installed=0"

if exist "%ProgramFiles(x86)%\Microsoft\EdgeWebView\Application" (
    set "webview2_installed=1"
)


echo                                       [1/4] .NET Desktop Runtime 8

if "%dotnet8_installed%"=="1" (

    echo                                              JA INSTALADO

) else (

    echo                                              BAIXANDO...

    curl -f -L ^
    -o "%temp%\dotnet8.exe" ^
    "https://builds.dotnet.microsoft.com/dotnet/WindowsDesktop/8.0.27/windowsdesktop-runtime-8.0.27-win-x64.exe"

    if exist "%temp%\dotnet8.exe" (
        start /wait "" "%temp%\dotnet8.exe" /install /quiet /norestart
        del "%temp%\dotnet8.exe" /f /q >nul 2>&1
    )
)


echo.
echo                                       [2/4] Visual C++ x64

if "%vcredist64_installed%"=="1" (

    echo                                              JA INSTALADO

) else (

    echo                                              BAIXANDO...

    curl -f -L ^
    -o "%temp%\vc64.exe" ^
    "https://aka.ms/vc14/vc_redist.x64.exe"

    if exist "%temp%\vc64.exe" (
        start /wait "" "%temp%\vc64.exe" /install /quiet /norestart
        del "%temp%\vc64.exe" /f /q >nul 2>&1
    )
)


echo.
echo                                       [3/4] Visual C++ x86

if "%vcredist86_installed%"=="1" (

    echo                                              JA INSTALADO

) else (

    echo                                              BAIXANDO...

    curl -f -L ^
    -o "%temp%\vc86.exe" ^
    "https://aka.ms/vc14/vc_redist.x86.exe"

    if exist "%temp%\vc86.exe" (
        start /wait "" "%temp%\vc86.exe" /install /quiet /norestart
        del "%temp%\vc86.exe" /f /q >nul 2>&1
    )
)


echo.
echo                                       [4/4] WebView2

if "%webview2_installed%"=="1" (

    echo                                              JA INSTALADO

) else (

    echo                                              BAIXANDO...

    curl -f -L ^
    -o "%temp%\wv2.exe" ^
    "https://go.microsoft.com/fwlink/?linkid=2124701"

    if exist "%temp%\wv2.exe" (
        start /wait "" "%temp%\wv2.exe" /silent /install
        del "%temp%\wv2.exe" /f /q >nul 2>&1
    )
)

echo.
echo                                       Operacao concluida.
echo.
pause
goto menu

:: ============================================================
:: ROBLOX ACCOUNT MANAGER
:: ============================================================

:baixar_ram
cls
echo.
echo                    ============================================================================
echo.
echo                                      BAIXANDO ROBLOX ACCOUNT MANAGER
echo.
echo                    ============================================================================
echo.

set "ram_zip=%temp%\RobloxAccountManager.zip"
set "ram_dir=%USERPROFILE%\Downloads\Roblox Account Manager"

if exist "%ram_zip%" (
    del "%ram_zip%" /f /q >nul 2>&1
)

echo                                       Procurando versao mais recente...
echo.

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
"$ErrorActionPreference='Stop'; ^
$r=Invoke-RestMethod 'https://api.github.com/repos/ic3w0lf22/Roblox-Account-Manager/releases/latest'; ^
$a=$r.assets | Where-Object { $_.name -like '*.zip' } | Select-Object -First 1; ^
if(-not $a){throw 'Nenhum ZIP encontrado'}; ^
Invoke-WebRequest -Uri $a.browser_download_url -OutFile '%ram_zip%'"

if not exist "%ram_zip%" (
    echo.
    echo                                       [!] Erro ao baixar Roblox Account Manager.
    echo.
    pause
    goto menu
)

echo.
echo                                       Extraindo arquivos...
echo.

if exist "%ram_dir%" (
    rmdir /S /Q "%ram_dir%" >nul 2>&1
)

mkdir "%ram_dir%" >nul 2>&1

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
"Expand-Archive -LiteralPath '%ram_zip%' -DestinationPath '%ram_dir%' -Force"

if errorlevel 1 (
    echo.
    echo                                       [!] Erro ao extrair.
    echo.
    pause
    goto menu
)

del "%ram_zip%" /f /q >nul 2>&1

echo.
echo                                       Download concluido.
echo.
echo                                       Pasta:
echo                                       %ram_dir%
echo.

start "" "%ram_dir%"

pause
goto menu
