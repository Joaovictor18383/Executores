:: ============================================================================
:: FISHSTRAP
:: ============================================================================

:install_fishstrap

set "dotnet6_installed=0"

where dotnet >nul 2>&1

if not errorlevel 1 (
    dotnet --list-runtimes 2>nul | findstr /i "Microsoft.WindowsDesktop.App 6." >nul 2>&1
    if not errorlevel 1 set "dotnet6_installed=1"
)

echo.
echo                              [ 1 / 2 ] .NET Desktop Runtime 6
echo.

if "!dotnet6_installed!"=="1" (
    echo                              [ OK ] .NET 6 ja esta instalado.
) else (
    set "dotnet6_installer=%TEMP%\dotnet6_desktop.exe"

    if exist "!dotnet6_installer!" del /f /q "!dotnet6_installer!" >nul 2>&1

    echo                              [ .. ] Baixando .NET 6...
    echo.

    curl.exe -f -L --retry 3 --retry-delay 2 -o "!dotnet6_installer!" "%DOTNET6_URL%"

    if errorlevel 1 (
        echo.
        echo                              [ ERRO ] Falha ao baixar .NET 6.
        echo.
        pause
        exit /b
    )

    echo.
    echo                              [ .. ] Instalando .NET 6...
    echo.

    start /wait "" "!dotnet6_installer!" /install /quiet /norestart

    del /f /q "!dotnet6_installer!" >nul 2>&1
)

echo.
echo                              [ 2 / 2 ] Fishstrap
echo.

set "fishstrap_exe=%USERPROFILE%\Downloads\Fishstrap.exe"

if exist "%fishstrap_exe%" (
    taskkill /f /im Fishstrap.exe >nul 2>&1
    timeout /t 1 /nobreak >nul
    del /f /q "%fishstrap_exe%" >nul 2>&1
)

echo                              [ .. ] Baixando Fishstrap...
echo.

curl.exe -f -L --retry 3 --retry-delay 2 -o "%fishstrap_exe%" "%FISHSTRAP_URL%"

if errorlevel 1 (
    echo.
    echo                              [ ERRO ] Falha ao baixar Fishstrap.
    echo.
    pause
    exit /b
)

echo.
echo                              [ OK ] Fishstrap baixado.
echo.
echo                              [ .. ] Executando Fishstrap...
echo.

start "" "%fishstrap_exe%"

echo.
echo                              [ OK ] Fishstrap iniciado.
echo.

pause
exit /b


:: ============================================================================
:: ROBLOX ACCOUNT MANAGER
:: ============================================================================

:install_ram

set "ram_zip=%TEMP%\RobloxAccountManager.zip"
set "ram_dir=%USERPROFILE%\Downloads\Roblox Account Manager"

if exist "%ram_zip%" del /f /q "%ram_zip%" >nul 2>&1

echo.
echo                              [ .. ] Procurando versao mais recente...
echo.

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
"$ErrorActionPreference='Stop';" ^
"$r=Invoke-RestMethod 'https://api.github.com/repos/ic3w0lf22/Roblox-Account-Manager/releases/latest';" ^
"$a=$r.assets ^| Where-Object {$_.name -like '*.zip'} ^| Select-Object -First 1;" ^
"if(-not $a){throw 'ZIP nao encontrado'};" ^
"Invoke-WebRequest -UseBasicParsing -Uri $a.browser_download_url -OutFile '%ram_zip%'"

if errorlevel 1 (
    echo.
    echo                              [ ERRO ] Falha ao baixar Roblox Account Manager.
    echo.
    pause
    exit /b
)

if exist "%ram_dir%" rmdir /s /q "%ram_dir%" >nul 2>&1

mkdir "%ram_dir%" >nul 2>&1

echo.
echo                              [ .. ] Extraindo...
echo.

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
"Expand-Archive -LiteralPath '%ram_zip%' -DestinationPath '%ram_dir%' -Force"

if errorlevel 1 (
    echo.
    echo                              [ ERRO ] Falha ao extrair.
    echo.
    pause
    exit /b
)

del /f /q "%ram_zip%" >nul 2>&1

echo.
echo                              [ OK ] Roblox Account Manager baixado.
echo.
echo                              %ram_dir%
echo.

pause
exit /b
